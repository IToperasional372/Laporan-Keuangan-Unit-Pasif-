/** * REKAP LAPORAN KEUANGAN UNIT PASIF - MASTER AGREGATOR * Backend Google Apps Script (VERSI FINAL - ACCURATE & DYNAMIC) */ const SPREADSHEET_ID = '17w2Wi24WNunMjGBk7EA8vrvVcRLqWAGyTI0_xyIoVhQ'; const APP_USERNAME = 'Arifjuna'; const APP_PASSWORD = 'adminpusat'; function doGet() { return HtmlService.createHtmlOutputFromFile('Index') .setTitle('Rekap Master Unit Pasif') .addMetaTag('viewport', 'width=device-width, initial-scale=1') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } function checkLogin(username, password) { if (username === APP_USERNAME && password === APP_PASSWORD) { return { success: true, message: "Login berhasil!" }; } else { return { success: false, message: "Username atau Password salah!" }; } } // ========================================== // LOGIKA AGREGASI SHEET CABANG (ACCURATE & DYNAMIC) // ========================================== function getRekapMasterData(selectedUnit = 'Semua Unit', forceRefresh = false) { try { const cache = CacheService.getScriptCache(); const cacheKey = 'REKAP_DATA_V3_' + selectedUnit.replace(/\s+/g, '_'); // CEK MEMORI: Jika tidak dipaksa refresh, ambil data sekejap dari memori if (!forceRefresh) { const cachedData = cache.get(cacheKey); if (cachedData) { return JSON.parse(cachedData); } } const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const sheets = ss.getSheets(); // 1. DAFTAR SHEET PENGECUALIAN ADMIN const EXCLUDED_SHEETS = ['Sheet246', 'RANK', 'REKAPITULASI', 'Daftar_Sheet', 'Sheet1', 'Sheet2', 'Template']; // 2. NAMA-NAMA SHEET CABANG VALID const allTargetSheets = sheets .map(sheet => String(sheet.getName()).trim()) .filter(sheetName => !EXCLUDED_SHEETS.includes(sheetName) && sheetName !== ""); let targetSheets = allTargetSheets; if (selectedUnit && selectedUnit !== 'Semua Unit') { targetSheets = [selectedUnit]; } // 3. MENGAMBIL DAFTAR BULAN PERSIS SESUAI TAMPILAN SPREADSHEET (getDisplayValues) let masterMonths = []; for (let i = 0; i < sheets.length; i++) { let sName = String(sheets[i].getName()).trim(); if (!EXCLUDED_SHEETS.includes(sName) && sName !== "") { // Read Row 4 Display Values let headerDisplay = sheets[i].getRange(4, 1, 1, 35).getDisplayValues()[0]; // Cari kolom mana bulan pertama dimulai (abaikan kosong, 'bulan', 'kategori') let startCol = -1; for (let c = 0; c < 10; c++) { let val = String(headerDisplay[c] || "").trim().toLowerCase(); if (val !== "" && val !== "bulan" && val !== "kategori" && val !== "no" && val !== "kode") { startCol = c; break; } } if (startCol === -1) startCol = 2; // Fallback Col C for (let c = startCol; c < headerDisplay.length; c++) { let strVal = String(headerDisplay[c] || "").trim(); if (strVal !== "") { masterMonths.push(strVal); if (strVal.toLowerCase() === 'grand total' || strVal.toLowerCase() === 'total') break; } } break; // Ambil dari sheet valid pertama } } // Fallback jika tidak ditemukan if (masterMonths.length === 0) { masterMonths = ['Jul 25', 'Aug 25', 'Sep 25', 'Oct 25', 'Nov 25', 'Dec 25', 'Jan 26', 'Feb 26', 'Mar 26', 'Apr 26', 'May 26', 'Jun 26', 'Grand Total']; } // 4. STRUKTUR KATEGORI LENGKAP const structure = [ { title: 'DATA MURID', type: 'number', hasTotal: false, keys: ['Murid Trial Baru', 'Murid Baru', 'Murid Aktif Kembali', 'Murid Keluar', 'Murid Aktif', 'Pertumbuhan Murid', 'Murid Dhuafa', 'Murid Garansi', 'Jumlah Guru'] }, { title: 'PENERIMAAN', type: 'currency', hasTotal: true, keys: ['Daftar', 'SPP', 'Kaos', 'KPK', 'Sertifikat', 'STPB', 'Tas', 'BCABS', 'Event', 'Lain-Lain'] }, { title: 'PENGELUARAN', type: 'currency', hasTotal: true, keys: ['Voucher', 'Kaos', 'KPK', 'Sertifikat', 'STPB', 'Tas', 'BCABS', 'Modul', 'Imbalan', 'Komisi', 'Bonus Tahunan', 'Bagi Hasil', 'Sewa Tempat', 'Upah', 'Transportasi', 'Iuran & Sumbangan', 'Kegiatan', 'Telepon', 'Listrik', 'Air', 'ATK, ABM, FC & Fax', 'Rumah Tangga', 'Lain-Lain', 'Modul (Ongkir)', 'Perawatan'] }, { title: 'SETOR & KAS', type: 'currency', hasTotal: false, keys: ['Setoran', 'Saldo Kas'] }, { title: 'PRESENTASE %', type: 'percentage', hasTotal: false, keys: ['Biaya Operasional', 'Biaya Salary', 'Biaya Modul', 'Bagi Hasil', 'Sewa Tempat'] } ]; const aggData = {}; structure.forEach(sec => { sec.keys.forEach(key => { aggData[sec.title + "_" + key] = new Array(masterMonths.length).fill(0); }); }); let totalCabang = 0; const targetSheetsNormalized = targetSheets.map(s => String(s).trim().toLowerCase()); // 5. BACA & PETAKAN DATA DENGAN PEMETAAN NAMA BULAN sheets.forEach(sheet => { let sheetName = String(sheet.getName()).trim().toLowerCase(); if (!targetSheetsNormalized.includes(sheetName)) return; const lastRow = sheet.getLastRow(); if (lastRow < 5) return; // Header display row 4 untuk mapping kolom let headerDisplay = sheet.getRange(4, 1, 1, 35).getDisplayValues()[0]; let startCol = -1; for (let c = 0; c < 10; c++) { let val = String(headerDisplay[c] || "").trim().toLowerCase(); if (val !== "" && val !== "bulan" && val !== "kategori" && val !== "no" && val !== "kode") { startCol = c; break; } } if (startCol === -1) startCol = 2; // Buat mapping indeks kolom sheet -> indeks masterMonths let colToMasterIndex = {}; for (let c = startCol; c < headerDisplay.length; c++) { let mName = String(headerDisplay[c] || "").trim(); if (mName !== "") { let mIdx = masterMonths.indexOf(mName); if (mIdx !== -1) { colToMasterIndex[c] = mIdx; } if (mName.toLowerCase() === 'grand total' || mName.toLowerCase() === 'total') break; } } // Ambil data nilai const readRows = Math.min(lastRow, 150); const data = sheet.getRange(1, 1, readRows, headerDisplay.length).getValues(); totalCabang++; let currentSection = ""; for (let i = 0; i < data.length; i++) { // Cari label row di kolom 0, 1, 2, atau 3 let rowLabelStr = ""; for (let c = 0; c < Math.min(startCol + 1, 4); c++) { let valStr = String(data[i][c] || "").trim(); if (valStr !== "") { rowLabelStr = valStr; break; } } if (!rowLabelStr) continue; let upperRowStr = rowLabelStr.toUpperCase().replace(/\s+/g, ' '); let lowerRowStr = rowLabelStr.toLowerCase().replace(/\s+/g, ''); // Deteksi Perubahan Seksi if (upperRowStr.includes('DATA MURID')) { currentSection = 'DATA MURID'; continue; } else if (upperRowStr.includes('PENERIMAAN')) { currentSection = 'PENERIMAAN'; continue; } else if (upperRowStr.includes('PENGELUARAN')) { currentSection = 'PENGELUARAN'; continue; } else if (upperRowStr.includes('SETOR & KAS') || upperRowStr.includes('SETOR DAN KAS')) { currentSection = 'SETOR & KAS'; continue; } else if (upperRowStr.includes('PRESENTASE') || upperRowStr.includes('PERSENTASE')) { currentSection = 'PRESENTASE %'; continue; } if (currentSection) { let secConfig = structure.find(s => s.title === currentSection); if (secConfig) { let matchedKey = secConfig.keys.find(k => k.toLowerCase().replace(/\s+/g, '') === lowerRowStr); if (matchedKey) { let uniqueKey = currentSection + "_" + matchedKey; // Isi nilai ke bulan yang sesuai berdasarkan mapping kolom Object.keys(colToMasterIndex).forEach(c => { let colIdx = parseInt(c, 10); let masterIdx = colToMasterIndex[colIdx]; let cellVal = data[i][colIdx]; let val = 0; if (typeof cellVal === 'number') { val = cellVal; } else if (typeof cellVal === 'string') { let str = cellVal.trim(); if (str && str !== '-' && !str.includes('#VALUE!') && !str.includes('#DIV/0!')) { let isPct = str.includes('%'); let cleanStr = str.replace(/Rp/gi, '').replace(/\./g, '').replace(/,/g, '.').replace(/\s+/g, '').replace(/%/g, ''); let parsed = parseFloat(cleanStr); if (!isNaN(parsed)) { val = isPct ? (parsed / 100) : parsed; } } } aggData[uniqueKey][masterIdx] += val; }); } } } } }); // 6. HASIL AKHIR const sectionsResult = structure.map(sec => { let sectionRows = sec.keys.map(key => ({ name: key, values: aggData[sec.title + "_" + key] })); let totals = new Array(masterMonths.length).fill(0); if (sec.hasTotal) { sectionRows.forEach(row => { for(let i = 0; i < masterMonths.length; i++) totals[i] += row.values[i]; }); } return { title: sec.title, type: sec.type, hasTotal: sec.hasTotal, rows: sectionRows, totals: sec.hasTotal ? totals : [] }; }); const result = { success: true, months: masterMonths, sections: sectionsResult, cabang: totalCabang, selectedUnit: selectedUnit, branchList: allTargetSheets }; try { cache.put(cacheKey, JSON.stringify(result), 3600); } catch(e) {} return result; } catch (error) { return { success: false, error: error.toString() }; } } // ========================================== // DASHBOARD STATS // ========================================== function getDashboardStats(forceRefresh = false) { try { const cache = CacheService.getScriptCache(); const cacheKey = 'DASHBOARD_STATS_V3'; if (!forceRefresh) { const cachedData = cache.get(cacheKey); if (cachedData) { return JSON.parse(cachedData); } } const rekap = getRekapMasterData('Semua Unit', forceRefresh); if (!rekap.success) throw new Error("Gagal mengambil rekap"); let totalPenerimaan = 0; let totalPengeluaran = 0; const lastIdx = rekap.months.length - 1; rekap.sections.forEach(sec => { if (sec.title === 'PENERIMAAN') totalPenerimaan = sec.totals[lastIdx] || 0; if (sec.title === 'PENGELUARAN') totalPengeluaran = sec.totals[lastIdx] || 0; }); const result = { penerimaan: totalPenerimaan, pengeluaran: totalPengeluaran, saldo: totalPenerimaan - totalPengeluaran, cabang: rekap.cabang, branchList: rekap.branchList }; try { cache.put(cacheKey, JSON.stringify(result), 3600); } catch(e) {} return result; } catch(error) { return { penerimaan: 0, pengeluaran: 0, saldo: 0, cabang: 0, branchList: [] }; } }
